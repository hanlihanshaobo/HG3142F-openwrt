# ---- 网口调试（TFTP 加载 initramfs）----
ipaddr=192.168.1.10
serverip=192.168.1.2
netmask=255.255.255.0
loadaddr=0x01000000

# 阶段 1：网络启动，不碰 flash
tftpboot_initramfs=tftp ${loadaddr} openwrt-initramfs.bin; bootm ${loadaddr}
tftpboot_kernel=tftp ${loadaddr} zImage; tftp ${fdtaddr} bcm96855-mb4784w.dtb; bootz ${loadaddr} - ${fdtaddr}
